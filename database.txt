-- =========================================================================
-- KHỞI TẠO CƠ SỞ DỮ LIỆU READLN TRÊN MYSQL (CHUẨN NIST CSF 1.1 - KHÔNG ẢNH BÌA)
-- =========================================================================

DROP DATABASE IF EXISTS readln_db;
CREATE DATABASE readln_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE readln_db;

-- =========================================================================
-- 1. TÀI KHOẢN & BẢO MẬT (NIST: Identify & Protect)
-- =========================================================================
CREATE TABLE users (
    id VARCHAR(36) PRIMARY KEY, -- UUID v4
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL, -- Argon2id hoặc Bcrypt
    role ENUM('READER', 'AUTHOR', 'ADMIN') NOT NULL DEFAULT 'READER',
    failed_login_attempts INT NOT NULL DEFAULT 0, -- Chống Brute-force
    lockout_until DATETIME NULL, -- Khóa tạm thời sau 5 lần nhập sai
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE password_resets (
    id VARCHAR(36) PRIMARY KEY,
    user_id VARCHAR(36) NOT NULL,
    token_hash VARCHAR(255) NOT NULL, -- SHA-256 Token
    expires_at DATETIME NOT NULL,
    used TINYINT(1) NOT NULL DEFAULT 0,
    CONSTRAINT fk_reset_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- =========================================================================
-- 2. THỂ LOẠI & TRUYỆN (Đã bỏ cột ảnh bìa)
-- =========================================================================
CREATE TABLE genres (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE,
    slug VARCHAR(50) NOT NULL UNIQUE
) ENGINE=InnoDB;

CREATE TABLE stories (
    id VARCHAR(36) PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    slug VARCHAR(255) NOT NULL UNIQUE,
    author_name VARCHAR(100) NOT NULL,
    summary TEXT NULL,
    view_count BIGINT NOT NULL DEFAULT 0,
    is_deleted TINYINT(1) NOT NULL DEFAULT 0, -- NIST Recover: Soft Delete
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE story_genres (
    story_id VARCHAR(36) NOT NULL,
    genre_id INT NOT NULL,
    PRIMARY KEY (story_id, genre_id),
    CONSTRAINT fk_sg_story FOREIGN KEY (story_id) REFERENCES stories(id) ON DELETE CASCADE,
    CONSTRAINT fk_sg_genre FOREIGN KEY (genre_id) REFERENCES genres(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- =========================================================================
-- 3. CHƯƠNG TRUYỆN
-- =========================================================================
CREATE TABLE chapters (
    id VARCHAR(36) PRIMARY KEY,
    story_id VARCHAR(36) NOT NULL,
    chapter_number INT NOT NULL,
    title VARCHAR(255) NOT NULL,
    content MEDIUMTEXT NOT NULL, -- Chứa nội dung chương dung lượng lớn (tối đa 16MB)
    word_count INT NOT NULL DEFAULT 0,
    is_deleted TINYINT(1) NOT NULL DEFAULT 0, -- Soft Delete
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT uq_story_chapter UNIQUE (story_id, chapter_number),
    CONSTRAINT fk_chapter_story FOREIGN KEY (story_id) REFERENCES stories(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- =========================================================================
-- 4. TƯƠNG TÁC NGƯỜI DÙNG: LƯU TRUYỆN, ĐÁNH GIÁ, BÌNH LUẬN
-- =========================================================================
CREATE TABLE bookmarks (
    user_id VARCHAR(36) NOT NULL,
    story_id VARCHAR(36) NOT NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (user_id, story_id),
    CONSTRAINT fk_bm_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    CONSTRAINT fk_bm_story FOREIGN KEY (story_id) REFERENCES stories(id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE ratings (
    id VARCHAR(36) PRIMARY KEY,
    story_id VARCHAR(36) NOT NULL,
    user_id VARCHAR(36) NOT NULL,
    score TINYINT NOT NULL CHECK (score >= 1 AND score <= 5),
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT uq_story_user_rating UNIQUE (story_id, user_id),
    CONSTRAINT fk_rating_story FOREIGN KEY (story_id) REFERENCES stories(id) ON DELETE CASCADE,
    CONSTRAINT fk_rating_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE comments (
    id VARCHAR(36) PRIMARY KEY,
    story_id VARCHAR(36) NOT NULL,
    chapter_id VARCHAR(36) NULL,
    user_id VARCHAR(36) NOT NULL,
    content TEXT NOT NULL,
    is_flagged TINYINT(1) NOT NULL DEFAULT 0,
    is_deleted TINYINT(1) NOT NULL DEFAULT 0,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_comment_story FOREIGN KEY (story_id) REFERENCES stories(id) ON DELETE CASCADE,
    CONSTRAINT fk_comment_chapter FOREIGN KEY (chapter_id) REFERENCES chapters(id) ON DELETE SET NULL,
    CONSTRAINT fk_comment_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- =========================================================================
-- 5. BẢO MẬT & VẬN HÀNH NIST CSF 1.1
-- =========================================================================
-- [NIST: DETECT] Nhật ký an ninh (Audit Logs)
CREATE TABLE audit_logs (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id VARCHAR(36) NULL,
    client_ip VARCHAR(45) NOT NULL,
    user_agent TEXT NULL,
    action VARCHAR(100) NOT NULL,
    status_code INT NOT NULL,
    details JSON NULL, -- MySQL hỗ trợ kiểu JSON gốc rất mạnh
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- [NIST: RESPOND] Danh sách IP bị chặn
CREATE TABLE ip_blacklists (
    id INT AUTO_INCREMENT PRIMARY KEY,
    ip_address VARCHAR(45) NOT NULL UNIQUE,
    reason VARCHAR(255) NOT NULL,
    blocked_until DATETIME NULL, -- NULL = Cấm vĩnh viễn
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- [NIST: RESPOND] Cấu hình hệ thống & Nút khẩn cấp (Kill-Switch)
CREATE TABLE system_settings (
    setting_key VARCHAR(50) PRIMARY KEY,
    setting_value VARCHAR(255) NOT NULL,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- =========================================================================
-- 6. ĐÁNH CHỈ MỤC TỐI ƯU TRUY VẤN (INDEXING)
-- =========================================================================
CREATE INDEX idx_stories_updated_at ON stories(is_deleted, updated_at DESC);
CREATE INDEX idx_chapters_story_id ON chapters(story_id, is_deleted, chapter_number);
CREATE INDEX idx_comments_story_id ON comments(story_id, is_deleted, created_at DESC);
CREATE INDEX idx_audit_logs_created_at ON audit_logs(created_at DESC);
CREATE INDEX idx_audit_logs_client_ip ON audit_logs(client_ip);
